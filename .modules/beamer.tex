% Abort when loaded in a non-Beamer document.
\@ifclassloaded{beamer}{}{
    \PackageError{.modules/beamer.tex}{Loaded in a non-Beamer document}
}

% -----------------------------------------------------------------------------
% Theme and color

\usetheme{moloch}
\molochcolors{theme=paper}

% Adjust theme colors
\molochcolors{light/title separator=CardinalRed}

% No color for links
\hypersetup{allcolors = .}

% Use the `fill` style for blocks
\molochset{block=fill}

% -----------------------------------------------------------------------------
% Inner theme

% More spacing between paragraphs
\newlength{\BeamerParskip}
\setlength{\BeamerParskip}{1em plus 0.2em minus 0.2em} % default is 0pt
\setlength{\parskip}{\BeamerParskip}

% Special handling is needed for `columns` environments
\def\beamer@setcolparskip{\setlength{\parskip}{\BeamerParskip}}
\define@key{beamer@col}{c}[true]{\def\beamer@colmode{c}\def\beamer@colalign{c}%
    \def\beamer@colheadskip{\beamer@setcolparskip}}
\define@key{beamer@col}{t}[true]{\def\beamer@colmode{t}\def\beamer@colalign{t}%
    \def\beamer@colheadskip{\beamer@setcolparskip}}
\define@key{beamer@col}{T}[true]{\def\beamer@colmode{T}\def\beamer@colalign{t}%
    \def\beamer@colheadskip{\vskip-1ex\beamer@setcolparskip\vskip-\parskip\nointerlineskip}}
\define@key{beamer@col}{b}[true]{\def\beamer@colmode{b}\def\beamer@colalign{b}%
    \def\beamer@colheadskip{\beamer@setcolparskip}}

% Less spacing before lists
\newcommand{\reduceSpacingIfZeroDepth}{\ifnum\@itemdepth=0 \vspace{-1em}\fi}
\AddToHook{env/itemize/before}{\reduceSpacingIfZeroDepth}
\AddToHook{env/enumerate/before}{\reduceSpacingIfZeroDepth}

% -----------------------------------------------------------------------------
% Outer theme

% Add page numbers
\setbeamertemplate{page number in head/foot}[totalframenumber]

% -----------------------------------------------------------------------------
% Bibliography related

% Use "et al." for more than two authors in citations
\ExecuteBibliographyOptions{maxcitenames=2}

% Remove document icons in the bibliography
\setbeamertemplate{bibliography item}[text]

% Use "&" instead of "and" for in-text citations
\DeclareDelimFormat{finalnamedelim}{%
    \ifnumgreater{\value{liststop}}{2}{\finalandcomma}{}%
    \addspace\&\space}
% ... but set it back for the bibliography
\DeclareDelimFormat[bib,biblist]{finalnamedelim}{%
    \ifnumgreater{\value{liststop}}{2}{\finalandcomma}{}%
    \addspace\bibstring{and}\space}
