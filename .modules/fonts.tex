% Loading fonts by font family name is not supported by XeTeX on macOS. Loading
% fonts by file name requires specifying variants and features explicitly.
% Therefore, load OpenType fonts with dedicated packages whenever possible.

\newcommand{\sansFontScale}{0.93}
\newcommand{\monoFontScale}{0.93}

\ifpdftex

    % New PX (serif and math)
    \usepackage{newpx}

    % Fira Sans (sans-serif)
    \usepackage[scaled=\sansFontScale]{FiraSans}

    % Fira Code (monospace)
    \usepackage[scaled=\monoFontScale]{FiraMono}

\else

    \usepackage{fontspec}

    % TeX Gyre Pagella (text)
    \setmainfont{TeX Gyre Pagella}

    % Fira Sans (sans-serif)
    \usepackage[scale=\sansFontScale]{FiraSans}

    % JetBrains Mono (monospace)
    \IfFontExistsTF{JetBrains Mono}{
        \setmonofont{JetBrains Mono}[Scale=MatchLowercase]
    }{
        % JetBrains Mono is not part of TeX Live. Use Fira Mono if it is not
        % available.
        \usepackage[scale=\monoFontScale]{FiraMono}
    }

    % `unicode-math` must be loaded after `newpxtext`
    \usepackage[
        math-style=ISO, bold-style=ISO,
        partial=upright,
        mathrm=sym, mathit=sym, mathsf=sym, mathbf=sym, mathtt=sym,
        warnings-off={mathtools-colon,mathtools-overbracket},
    ]{unicode-math}

    % TeX Gyre Pagella (math)
    \setmathfont{texgyrepagella-math.otf}

    % Load New Computer Modern (book weight) for most symbols
    \setmathfont{NewCMMath-Book.otf}[
        range={
                \mathopen,\mathclose,\mathfence,\mathop,\mathbin,\mathrel,
                \mathcal,\varnothing
            },
    ]
    % ... but set some of them back to Pagella
    \setmathfont{texgyrepagella-math.otf}[
        range={\mathexclam}
    ]

    % Load Asana Math for letter `f` only (it has better kerning)
    \setmathfont{Asana-Math.otf}[range=\mitf]
    \begingroup\lccode`~=`f \lowercase{\endgroup\def~}{\mitf}
    \mathcode`f="8000
    % From <https://tex.stackexchange.com/a/530981>

    % LuaTeX reads the `MathConstants` table from the last loaded math font for
    % parameters such as the thickness of fraction rules. Here we load New
    % Computer Modern with an empty range to use its parameters. See
    % <https://tug.ctan.org/systems/doc/luatex/luatex-math.tex>. XeTeX handles
    % this differently. I am not sure how it works.
    \setmathfont{NewCMMath-Book.otf}[range={}]

\fi
