% Loading fonts by font family name is not supported by XeTeX on macOS. Loading
% fonts by file name requires specifying variants and features explicitly.
% Therefore, load OpenType fonts with dedicated packages whenever possible.

% Font scale
\newcommand{\FontScale}{1.2}
\newcommand{\FiraMonoScale}{1}

\ifpdftex
    % New PX (serif and math)
    \usepackage[scale=\FontScale]{newpx}
\else
    % unicode-math
    \usepackage[
        math-style=ISO, bold-style=ISO,
        partial=upright,
        mathrm=sym, mathit=sym, mathsf=sym, mathbf=sym, mathtt=sym,
        warnings-off={mathtools-colon,mathtools-overbracket},
    ]{unicode-math}

    % TeX Gyre Pagella (serif)
    \usepackage[
        ScaleRM=\FontScale,
    ]{pagella-otf}

    % Fira Sans (sans-serif)
    % \usepackage[
    %     scale=\FontScale,
    % ]{FiraSans}

    % Fira Mono (monospace)
    \usepackage[
        scale=\FiraMonoScale,
        nomap,
    ]{FiraMono}

    % TeX Gyre Pagella (math)
    \setmathfont{texgyrepagella-math.otf}[
        Scale=\FontScale,
        AutoFakeBold,
    ]
    % Package `pagella-otf` also loads the math font, but it does not scale it
    % according to its `ScaleRM` option. We must load it again.

    % Load Asana Math for letter `f` only (it has better kerning)
    \setmathfont{Asana-Math.otf}[
        Scale=\FontScale,
        AutoFakeBold,
        range=\mitf,
    ]

    % Map letter `f` to `\mitf` in math mode
    \begingroup\lccode`~=`f \lowercase{\endgroup\def~}{\mitf}
    \mathcode`f="8000
    % From <https://tex.stackexchange.com/a/530981>

    % Load New Computer Modern for select symbols
    \setmathfont{NewCMMath-Book.otf}[ % use book weight
        Scale=\FontScale,
        AutoFakeBold,
        range={
                \mathfence,\mathop,\mathbin,\mathrel,
                \sqrt,\mathcal,\varnothing,
            },
    ]
\fi
