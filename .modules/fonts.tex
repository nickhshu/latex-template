\newcommand{\firaSansScale}{0.93}
\newcommand{\firaMonoScale}{0.93}

\ifluatex

    \usepackage{fontspec}
    \usepackage[
        math-style=ISO, bold-style=ISO,
        partial=upright,
        mathrm=sym, mathit=sym, mathsf=sym, mathbf=sym, mathtt=sym,
        warnings-off={mathtools-colon,mathtools-overbracket},
    ]{unicode-math}

    % TeX Gyre Pagella (text)
    \setmainfont{TeX Gyre Pagella}

    % Fira Sans (sans-serif)
    \setsansfont{Fira Sans}[Scale=\firaSansScale]

    % JetBrains Mono (monospace)
    \IfFontExistsTF{JetBrains Mono}{
        \setmonofont{JetBrains Mono}[Scale=MatchLowercase]
    }{
        % JetBrains Mono is not part of TeX Live. Use Fira Mono if it is not
        % available.
        \setmonofont{Fira Mono}[Scale=\firaMonoScale]
    }

    % The math font is a blend of Pagella and New Computer Modern (book weight)
    \setmathfont{TeX Gyre Pagella Math}
    \setmathfont{NewCMMath-Book.otf}[
        range={
                \mathopen,\mathclose,\mathfence,\mathop,\mathbin,\mathrel,
                \mathcal,\varnothing
            },
    ]
    \setmathfont{TeX Gyre Pagella Math}[range={\mathexclam}]
    \setmathfont{NewCMMath-Book.otf}[range={}]
    % LuaTeX reads the `MathConstants` table from the last loaded math font for
    % parameters such as the thickness of fraction rules. Here we load New
    % Computer Modern with an empty range to use its parameters. See
    % <https://tug.ctan.org/systems/doc/luatex/luatex-math.tex>.

\else

    % New PX (serif and math)
    \usepackage{newpx}

    % Fira Sans (sans-serif)
    \usepackage[scale=\firaSansScale]{FiraSans}

    % Fira Code (monospace)
    \usepackage[scale=\firaMonoScale]{FiraMono}

\fi
