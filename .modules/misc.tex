% -----------------------------------------------------------------------------
% Page layout and paragraph formatting

% A4 paper with vertical margins slightly smaller than default (0.7)
\@ifclassloaded{article}{
    \usepackage[a4paper, vscale=0.75]{geometry}
}

% Line spacing slightly smaller than one-half spacing (1.25)
\usepackage{setspace}
\setstretch{1.2}
% Note that this applies to Beamer slides as well.

% No first-paragraph indentation for quotations and abstracts (which uses the
% `quotation` environment)
\appto\quotation{\noindent\ignorespaces}

% -----------------------------------------------------------------------------
% Sections

% Add a dot after section numbers
\renewcommand*{\@seccntformat}[1]{\csname the#1\endcsname.\quad}
% See <https://texfaq.org/FAQ-seccntfmt>.

% Also update `hyperref` bookmarks
\AtEndPreamble{
    % Postpone until `hyperref` is loaded at the end of the preamble
    \renewcommand\Hy@numberline[1]{#1. }
}
% From <https://tex.stackexchange.com/a/557745>.

% Section with subtitles
\NewDocumentCommand{\TwoPartSection}{o m m}{%
    \IfNoValueTF{#1}
    {\section[#2]{#2\\\textmd{\textit{\large #3}}}}%
    {\section[#1]{#2\\\textmd{\textit{\large #3}}}}%
}
\NewDocumentCommand{\TwoPartSubsection}{o m m}{%
    \IfNoValueTF{#1}
    {\subsection[#2]{#2\\\textmd{\textit{\large #3}}}}%
    {\subsection[#1]{#2\\\textmd{\textit{\large #3}}}}%
}

% -----------------------------------------------------------------------------
% Appendices

% Appendix sections will appear as
% - "Appendix 1. Title" in text,
% - "A1. Title" in PDF bookmarks, and
% - "Appendix 1" when cited with `\cref`.

% Appendix subsections (and subsubsections) will appear as
% - "A1.1. Title" in text and PDF bookmarks and
% - "Section A1.1" when cited with `\cref`.

\newcommand{\appfmt@seccntformat}[1]{\csname appfmt@#1\endcsname}
\newcommand{\appfmt@section}{Appendix~\thesection.\quad}
\newcommand{\appfmt@subsection}{\thesubsection.\quad}
\newcommand{\appfmt@subsubsection}{\thesubsubsection.\quad}
\newcommand{\appfmt@numberline}[1]{%
    \ifnum\pdf@strcmp{#1}{\thesection}=0 A\fi #1. }

\AddToHook{cmd/appendix/after}{%
    \renewcommand{\thesection}{\arabic{section}}%
    \renewcommand{\thesubsection}{A\thesection.\arabic{subsection}}%
    \let\@seccntformat\appfmt@seccntformat
    \let\Hy@numberline\appfmt@numberline
    \zcsetup{countertype={section=appendix, subsection=section, subsubsection=section}}%
}

% -----------------------------------------------------------------------------
% Text formatting

% Micro-typographic refinements
\usepackage{microtype}

% Smart quotation marks
\usepackage[maxlevel=4]{csquotes}
% Load before `biblatex-chicago'

% Fix spacing issues for ellipses
\usepackage{ellipsis}
% See the Companion, 3rd ed., vol. 1, p. 148.

% Commas between consecutive footnotes
\AtEndPreamble{
    % Postpone until `hyperref` is loaded at the end of the preamble
    \let\oldFootnote\footnote
    \newcommand\nextToken\relax
    \renewcommand\footnote[1]{\oldFootnote{#1}\futurelet\nextToken\isFootnote}
    \newcommand\isFootnote{\ifx\footnote\nextToken\textsuperscript{,}\fi}
}
% From <https://tex.stackexchange.com/a/62091>.
